-- Esquema canónico para Pizzería Bella Italia (referencia)
-- Ejecutar en SQL Server / LocalDB. Requiere permisos para crear base de datos.

IF DB_ID(N'pizzeriaBD') IS NULL
    CREATE DATABASE pizzeriaBD;
GO

USE pizzeriaBD;
GO

IF OBJECT_ID(N'dbo.DetallePedidos', N'U') IS NOT NULL DROP TABLE dbo.DetallePedidos;
IF OBJECT_ID(N'dbo.Pedidos', N'U') IS NOT NULL DROP TABLE dbo.Pedidos;
IF OBJECT_ID(N'dbo.Clientes', N'U') IS NOT NULL DROP TABLE dbo.Clientes;
GO

CREATE TABLE dbo.Clientes (
    IdCliente INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    Telefono NVARCHAR(30) NOT NULL
);

CREATE TABLE dbo.Pedidos (
    IdPedido INT IDENTITY(1,1) PRIMARY KEY,
    IdCliente INT NOT NULL,
    FechaHora DATETIME2(0) NOT NULL DEFAULT SYSDATETIME(),
    EsDelivery BIT NOT NULL DEFAULT 0,
    AplicaConvenioEstudiante BIT NOT NULL DEFAULT 0,
    Subtotal DECIMAL(18,2) NOT NULL DEFAULT 0,
    Descuento DECIMAL(18,2) NOT NULL DEFAULT 0,
    DeliveryCosto DECIMAL(18,2) NOT NULL DEFAULT 0,
    Iva DECIMAL(18,2) NOT NULL DEFAULT 0,
    Total DECIMAL(18,2) NOT NULL DEFAULT 0,
    CONSTRAINT FK_Pedidos_Clientes FOREIGN KEY (IdCliente) REFERENCES dbo.Clientes(IdCliente)
);

CREATE TABLE dbo.DetallePedidos (
    IdDetalle INT IDENTITY(1,1) PRIMARY KEY,
    IdPedido INT NOT NULL,
    Tamano NVARCHAR(50) NOT NULL,
    TipoMasa NVARCHAR(50) NOT NULL,
    QuesoExtra BIT NOT NULL DEFAULT 0,
    Pepperoni BIT NOT NULL DEFAULT 0,
    Champinones BIT NOT NULL DEFAULT 0,
    Aceitunas BIT NOT NULL DEFAULT 0,
    Tocino BIT NOT NULL DEFAULT 0,
    Cantidad INT NOT NULL DEFAULT 1,
    PrecioUnitario DECIMAL(18,2) NOT NULL DEFAULT 0,
    SubtotalItem DECIMAL(18,2) NOT NULL DEFAULT 0,
    Resumen NVARCHAR(300) NOT NULL DEFAULT N'',
    CONSTRAINT FK_DetallePedidos_Pedidos FOREIGN KEY (IdPedido) REFERENCES dbo.Pedidos(IdPedido) ON DELETE CASCADE
);
GO
